create table if not exists fupinzhunong.chat
(
    id              int auto_increment comment '主键'
        primary key,
    yonghu_id       int          null comment '提问用户',
    chat_issue      varchar(200) null comment '问题',
    issue_time      timestamp    null comment '问题时间 Search111',
    chat_reply      varchar(200) null comment '回复',
    reply_time      timestamp    null comment '回复时间 Search111',
    zhuangtai_types int(255)     null comment '状态',
    chat_types      int          null comment '数据类型',
    insert_time     timestamp    null comment '创建时间'
)
    comment '客服聊天' charset = utf8;
